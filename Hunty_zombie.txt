local function getCharacterParts()

local function setAntiFall(hrp, enabled)
    if not hrp or not hrp.Parent then return end
    local bv = hrp:FindFirstChild("HX_AntiFall")
    if enabled then
        if not bv then
            bv = Instance.new("BodyVelocity")
            bv.Name = "HX_AntiFall"
            bv.MaxForce = Vector3.new(1e6, 1e6, 1e6)
            bv.Velocity = Vector3.zero
            bv.Parent = hrp
        end
    elseif bv then
        bv:Destroy()
    end
end

local function setCharacterNoclip(char, enabled)
    if not char then return end
    if enabled then
        IsNoclipApplied = true
        for _, part in ipairs(char:GetDescendants()) do
            if part:IsA("BasePart") then
                if OriginalCollideCache[part] == nil then
                    OriginalCollideCache[part] = part.CanCollide
                end
                if part.CanCollide then part.CanCollide = false end
            end
        end
    elseif IsNoclipApplied then
        IsNoclipApplied = false
        for part, was in pairs(OriginalCollideCache) do
            if part and part.Parent then part.CanCollide = was end
        end
        table.clear(OriginalCollideCache)
    end
end

local DoorCooldowns = {}
local LastDoorsFolder = nil

local function findDoorsFolder()
    for _, child in ipairs(Workspace:GetChildren()) do
        if child:IsA("Model") or child:IsA("Folder") then
            local doors = child:FindFirstChild("Doors")
            if doors then return doors end
        end
    end
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if (obj:IsA("Folder") or obj:IsA("Model")) and string.lower(obj.Name) == "doors" then
            return obj
        end
    end
    return nil
end

local function getGateStandPosition(gate, myPos)
    local cf, size
    if gate:IsA("Model") then
        local ok, c, s = pcall(function() return gate:GetBoundingBox() end)
        if not ok then return nil end
        cf, size = c, s
    elseif gate:IsA("BasePart") then
        cf, size = gate.CFrame, gate.Size
    else
        return nil
    end
    local bottomY = cf.Position.Y - (size.Y * 0.5) + 3
    local safeY = math.clamp(bottomY, myPos.Y - 6, myPos.Y + 10)
    local toMe = Vector3.new(myPos.X - cf.Position.X, 0, myPos.Z - cf.Position.Z)
    local offset = toMe.Magnitude > 0.5 and (toMe.Unit * 3.5) or Vector3.new(0, 0, 3.5)
    return Vector3.new(cf.Position.X + offset.X, safeY, cf.Position.Z + offset.Z)
end

local function getAvailableDoors(doorsFolder)
    if not doorsFolder then return {} end
    if doorsFolder ~= LastDoorsFolder then
        table.clear(DoorCooldowns)
        LastDoorsFolder = doorsFolder
    end
    local now = os.clock()
    local available = {}
    for _, door in ipairs(doorsFolder:GetChildren()) do
        local isLocked = door:GetAttribute("locked") == true
        local cdUntil = DoorCooldowns[door] or 0
        if not isLocked and now >= cdUntil then table.insert(available, door) end
    end
    table.sort(available, function(a, b)
        return (a:GetAttribute("roomIndex") or 999) < (b:GetAttribute("roomIndex") or 999)
    end)
    return available
end

local function processAutoDoors(hrp)
    local doorsFolder = findDoorsFolder()
    local doors = getAvailableDoors(doorsFolder)
    if #doors == 0 then return false end
    local door = doors[1]
    if not door or door.Parent ~= doorsFolder then return false end
    local standPos = getGateStandPosition(door, hrp.Position)
    if not standPos then
        DoorCooldowns[door] = os.clock() + 4
        return false
    end
    setAntiFall(hrp, true)
    for attempt = 1, 3 do
        if not isAliveRun() or not State.AutoDoors or not hrp.Parent then return true end
        if door.Parent ~= doorsFolder then return true end
        local targetPos = standPos
        if attempt == 2 then
            local cf = door:IsA("Model") and select(1, door:GetBoundingBox()) or door.CFrame
            targetPos = Vector3.new(cf.Position.X, standPos.Y, cf.Position.Z)
        end
        hrp.CFrame = CFrame.new(targetPos)
        hrp.AssemblyLinearVelocity = Vector3.zero
        task.wait(0.15)
    end
    if door.Parent == doorsFolder then DoorCooldowns[door] = os.clock() + 4 end
    return true
end

local MinSafeEnemyY = -20

local function getEnemyRoot(entity)
    local hrp = entity:FindFirstChild("HumanoidRootPart") or entity:FindFirstChild("HumanoidRootPart", true)
    if hrp and hrp:IsA("BasePart") then return hrp end
    local head = entity:FindFirstChild("Head") or entity:FindFirstChild("Head", true)
    if head and head:IsA("BasePart") then return head end
    if entity:IsA("Model") and entity.PrimaryPart then return entity.PrimaryPart end
    return entity:FindFirstChildWhichIsA("BasePart", true)
end

local function isEntityAlive(entity)
    if not entity or not entity.Parent then return false end
    if entity == LocalPlayer.Character then return false end
    if Players:GetPlayerFromCharacter(entity) then return false end
    local entitiesFolder = Workspace:FindFirstChild("Entities")
    if not entitiesFolder or not entity:IsDescendantOf(entitiesFolder) then return false end
    local hum = entity:FindFirstChildOfClass("Humanoid")
    if hum and (hum.Health <= 0 or hum:GetState() == Enum.HumanoidStateType.Dead) then return false end
    local hpAttr = entity:GetAttribute("Health") or entity:GetAttribute("HP")
        or entity:GetAttribute("CurrentHealth") or entity:GetAttribute("hp")
    if type(hpAttr) == "number" and hpAttr <= 0 then return false end
    if entity:GetAttribute("Dead") == true or entity:GetAttribute("IsDead") == true
        or entity:GetAttribute("dead") == true or entity:GetAttribute("Alive") == false then
        return false
    end
    local stateAttr = entity:GetAttribute("State") or entity:GetAttribute("Status")
    if type(stateAttr) == "string" then
        local l = string.lower(stateAttr)
        if string.find(l, "dead", 1, true) or string.find(l, "dying", 1, true) then return false end
    end
    for _, n in ipairs({"Health", "HP", "CurrentHealth"}) do
        local v = entity:FindFirstChild(n)
        if v and (v:IsA("NumberValue") or v:IsA("IntValue")) and v.Value <= 0 then return false end
    end
    for _, n in ipairs({"Dead", "IsDead"}) do
        local v = entity:FindFirstChild(n)
        if v and v:IsA("BoolValue") and v.Value == true then return false end
    end
    local head = entity:FindFirstChild("Head")
    if head and head:IsA("BasePart") and head.Transparency >= 0.9 and not State.EntityVisualHider then
        return false
    end
    for _, desc in ipairs(entity:GetDescendants()) do
        if desc:IsA("BillboardGui") then
            for _, ui in ipairs(desc:GetDescendants()) do
                if ui:IsA("TextLabel") then
                    local txt = string.gsub(ui.Text, "%s+", "")
                    if string.match(txt, "^0/") or txt == "0" or string.match(string.lower(txt), "^0hp") then
                        return false
                    end
                end
            end
        end
    end
    local root = getEnemyRoot(entity)
    if not root or not root.Parent then return false end
    if root.Position.Y < MinSafeEnemyY then return false end
    return true, root
end

local function findNearestLivingEnemy(myPos)
    local entitiesFolder = Workspace:FindFirstChild("Entities")
    if not entitiesFolder then return nil, nil end
    local bestEntity, bestRoot = nil, nil
    local shortest = math.max(100, State.ScanRange)
    local function evaluate(entity)
        local alive, root = isEntityAlive(entity)
        if not alive or not root then return end
        local d = (root.Position - myPos).Magnitude
        if d < shortest then shortest = d; bestEntity = entity; bestRoot = root end
    end
    for _, cat in ipairs(entitiesFolder:GetChildren()) do
        if cat:IsA("Folder") then
            for _, e in ipairs(cat:GetChildren()) do evaluate(e) end
        elseif cat:IsA("Model") then
            evaluate(cat)
        end
    end
    return bestEntity, bestRoot
end

local function computeAttackPosition(targetRoot)
    local dist = math.clamp(State.AttackDistance, 1, 25)
    local targetPos = targetRoot.Position
    if State.AttackType == "Above" then
        return targetPos + Vector3.new(0, math.max(dist, 0.5), 0), true
    elseif State.AttackType == "Below" then
        return targetPos + Vector3.new(0, -math.max(dist, 0.5), 0), true
    else
        local pos = (targetRoot.CFrame * CFrame.new(0, 0, dist)).Position
        if (pos - targetPos).Magnitude < 0.1 then pos = targetPos end
        return pos, false
    end
end

local function facePosition(fromPos, targetPos, keepFlat)
    if keepFlat then
        return CFrame.lookAt(fromPos, Vector3.new(targetPos.X, fromPos.Y, targetPos.Z))
    end
    return CFrame.lookAt(fromPos, targetPos)
end

local RaidSpawnPos = Vector3.new(16, 5, -292)
local RaidDestPos = Vector3.new(20, 12, -134)
local RaidSpawnRadius = 25

local function isWaitingAtRaidSpawn(hrp)
    if not State.RaidAutoTeleport or not hrp then return false end
    return (hrp.Position - RaidSpawnPos).Magnitude <= RaidSpawnRadius
end

task.spawn(function()
    while isAliveRun() do
        CurrentTweenPos = nil
        local char, hrp, hum = getCharacterParts()
        if char and hrp and hum and hum.Health > 0 then
            if isWaitingAtRaidSpawn(hrp) then
                ActiveTargetEntity, ActiveTargetRoot = nil, nil
                setAntiFall(hrp, false)
                setCharacterNoclip(char, false)
                task.wait(0.2)
            else
                local busyWithDoors = false
                if State.AutoDoors then busyWithDoors = processAutoDoors(hrp) end
                if busyWithDoors then
                    ActiveTargetEntity, ActiveTargetRoot = nil, nil
                elseif State.AutoFight then
                    local stillAlive = ActiveTargetEntity and isEntityAlive(ActiveTargetEntity)
                    if not stillAlive then
                        ActiveTargetEntity, ActiveTargetRoot = findNearestLivingEnemy(hrp.Position)
                    else
                        local ent, root = findNearestLivingEnemy(hrp.Position)
                        if ent and root then
                            local cur = ActiveTargetRoot and (ActiveTargetRoot.Position - hrp.Position).Magnitude or math.huge
                            local nw = (root.Position - hrp.Position).Magnitude
                            if nw + 8 < cur then ActiveTargetEntity, ActiveTargetRoot = ent, root end
                        end
                    end
                    if not ActiveTargetEntity then
                        setAntiFall(hrp, false)
                        setCharacterNoclip(char, false)
                    end
                else
                    ActiveTargetEntity, ActiveTargetRoot = nil, nil
                    setAntiFall(hrp, false)
                    setCharacterNoclip(char, false)
                end
            end
        else
            ActiveTargetEntity, ActiveTargetRoot = nil, nil
            if hrp then setAntiFall(hrp, false) end
            if char then setCharacterNoclip(char, false) end
        end
        task.wait(0.1)
    end
end)

trackConn(RunService.Heartbeat:Connect(function(dt)
    if not isAliveRun() or not State.AutoFight then return end
    local char, hrp, hum = getCharacterParts()
    if not char or not hrp or not hum or hum.Health <= 0 then return end
    if isWaitingAtRaidSpawn(hrp) then return end
    if not (ActiveTargetEntity and ActiveTargetRoot) then return end

    if not isEntityAlive(ActiveTargetEntity) or not ActiveTargetRoot.Parent then
        ActiveTargetEntity, ActiveTargetRoot = findNearestLivingEnemy(hrp.Position)
        if not (ActiveTargetEntity and ActiveTargetRoot) then
            setAntiFall(hrp, false)
            setCharacterNoclip(char, false)
            return
        end
    end

    local targetPos, isVertical = computeAttackPosition(ActiveTargetRoot)
    local mode = State.AttackMode

    if mode == "Teleport" then
        setAntiFall(hrp, true)
        setCharacterNoclip(char, State.AttackType == "Below")
        hrp.CFrame = facePosition(targetPos, ActiveTargetRoot.Position, not isVertical)
        hrp.AssemblyLinearVelocity = Vector3.zero
        hrp.AssemblyAngularVelocity = Vector3.zero
    elseif mode == "Tween" then
        setAntiFall(hrp, true)
        setCharacterNoclip(char, true)
        if not CurrentTweenPos then CurrentTweenPos = hrp.Position end
        local alpha = math.clamp(dt * 9, 0, 1)
        CurrentTweenPos = CurrentTweenPos:Lerp(targetPos, alpha)
        hrp.CFrame = facePosition(CurrentTweenPos, ActiveTargetRoot.Position, not isVertical)
        hrp.AssemblyLinearVelocity = Vector3.zero
    else
        setAntiFall(hrp, false)
        setCharacterNoclip(char, false)
        local flat = Vector3.new(targetPos.X, hrp.Position.Y, targetPos.Z)
        hrp.CFrame = facePosition(hrp.Position, Vector3.new(flat.X, hrp.Position.Y, flat.Z), true)
        hum:MoveTo(flat)
    end
end))

local function canCastCombat()
    if not State.OnlyCastWithTarget then return true end
    return ActiveTargetEntity ~= nil and ActiveTargetRoot ~= nil
end

local function skillReady(name)
    return os.clock() >= (SkillReadyAt[name] or 0)
end

local function markSkill(name, cd)
    SkillReadyAt[name] = os.clock() + (cd or State.SkillCooldown) / 1000
end

local function pressKey(keyCode)
    pcall(function()
        VirtualInputManager:SendKeyEvent(true, keyCode, false, game)
        task.wait(0.03)
        VirtualInputManager:SendKeyEvent(false, keyCode, false, game)
    end)
end

    while isAliveRun() do
        if State.AutoM1 and canCastCombat() and skillReady("M1") then
            markSkill("M1", State.M1Delay)
            pcall(function()
                local cam = Workspace.CurrentCamera
                local vp = cam and cam.ViewportSize or Vector2.new(500, 500)
                local cx, cy = vp.X * 0.5, vp.Y * 0.5
                VirtualInputManager:SendMouseButtonEvent(cx, cy, 0, true, game, 1)
                task.wait(0.02)
                VirtualInputManager:SendMouseButtonEvent(cx, cy, 0, false, game, 1)
            end)
            task.wait(State.M1Delay / 1000)
        else
            task.wait(0.05)
        end
    end
end)

local StatsGui = nil
local function setStatsTracker(enabled)
    if StatsGui then StatsGui:Destroy(); StatsGui = nil end
    if not enabled then return end

    local sg = Instance.new("ScreenGui")
    sg.Name = "HX_FloatingStats"
    sg.ResetOnSpawn = false
    sg.DisplayOrder = 100
    sg.Parent = getGuiParent()
    StatsGui = sg
    table.insert(CreatedGuis, sg)

    local card = Instance.new("Frame")
    card.Size = UDim2.new(0, 210, 0, 56)
    card.Position = UDim2.new(0.5, -105, 0, 12)
    card.BackgroundColor3 = Color3.fromRGB(20, 22, 30)
    card.BackgroundTransparency = 0.15
    card.Active = true
    card.Parent = sg
    Instance.new("UICorner", card).CornerRadius = UDim.new(0, 8)

    local stroke = Instance.new("UIStroke")
    stroke.Color = Color3.fromRGB(70, 80, 110)
    stroke.Thickness = 1
    stroke.Parent = card

    local title = Instance.new("TextLabel")
    title.Size = UDim2.new(1, -12, 0, 18)
    title.Position = UDim2.new(0, 6, 0, 4)
    title.BackgroundTransparency = 1
    title.Font = Enum.Font.Code
    title.TextSize = 11
    title.TextColor3 = Color3.fromRGB(160, 195, 255)
    title.TextXAlignment = Enum.TextXAlignment.Left
    title.Text = "Hyperium Stats"
    title.Parent = card

    local statsLabel = Instance.new("TextLabel")
    statsLabel.Size = UDim2.new(1, -12, 0, 30)
    statsLabel.Position = UDim2.new(0, 6, 0, 22)
    statsLabel.BackgroundTransparency = 1
    statsLabel.Font = Enum.Font.Code
    statsLabel.TextSize = 11
    statsLabel.TextColor3 = Color3.fromRGB(230, 240, 235)
    statsLabel.TextXAlignment = Enum.TextXAlignment.Left
    statsLabel.TextYAlignment = Enum.TextYAlignment.Top
    statsLabel.Text = "FPS: -- | Ping: --ms\nTime: 00:00:00"
    statsLabel.Parent = card

    local dragging, dragInput, dragStart, startPos
    card.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = true
            dragStart = input.Position
            startPos = card.Position
            dragInput = (input.UserInputType == Enum.UserInputType.Touch) and input or nil
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                    if dragInput == input then dragInput = nil end
                end
            end)
        end
    end)
    trackConn(UserInputService.InputChanged:Connect(function(input)
        if not dragging or not card.Parent then return end
        if input == dragInput or (dragInput == nil and input.UserInputType == Enum.UserInputType.MouseMovement) then
            local delta = input.Position - dragStart
            card.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end))

    local frameCount = 0
    local lastFpsTime = os.clock()
    local currentFps = 60

    local renderConn
    renderConn = RunService.RenderStepped:Connect(function()
        if not isAliveRun() or not StatsGui or not StatsGui.Parent then
            renderConn:Disconnect()
            return
        end
        frameCount += 1
        local now = os.clock()
        if now - lastFpsTime >= 0.5 then
            currentFps = math.floor((frameCount / (now - lastFpsTime)) + 0.5)
            frameCount = 0
            lastFpsTime = now
            local ping = 0
            pcall(function()
                ping = math.floor(Stats.Network.ServerStatsItem["Data Ping"]:GetValue())
            end)
            local elapsed = math.floor(now - SessionStartTime)
            local hrs = math.floor(elapsed / 3600)
            local mins = math.floor((elapsed % 3600) / 60)
            local secs = elapsed % 60
            statsLabel.Text = string.format("FPS: %d | Ping: %dms\nTime: %02d:%02d:%02d", currentFps, ping, hrs, mins, secs)
        end
    end)
    trackConn(renderConn)
end

local function applyFPSBooster(enabled)
    if enabled then
        if next(BoosterSavedEffects) then return end
        pcall(function()
            BoosterSavedEffects.__shadow = Lighting.GlobalShadows
            BoosterSavedEffects.__fog = Lighting.FogEnd
            BoosterSavedEffects.__quality = settings().Rendering.QualityLevel

            Lighting.GlobalShadows = false
            Lighting.FogEnd = 9e9
            settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
        end)

        for _, obj in ipairs(Lighting:GetChildren()) do
            if obj:IsA("PostEffect") or obj:IsA("Atmosphere") or obj:IsA("Sky") then
                if BoosterSavedEffects[obj] == nil then BoosterSavedEffects[obj] = obj.Enabled end
                obj.Enabled = false
            end
        end

        for _, obj in ipairs(Workspace:GetDescendants()) do
            if obj:IsA("ParticleEmitter") or obj:IsA("Trail") or obj:IsA("Beam")
                or obj:IsA("Fire") or obj:IsA("Smoke") or obj:IsA("Sparkles") then
                if BoosterSavedEffects[obj] == nil then BoosterSavedEffects[obj] = obj.Enabled end
                obj.Enabled = false
            elseif obj:IsA("BasePart") then
                pcall(function() obj.CastShadow = false end)
            end
        end
    else
        pcall(function()
            if BoosterSavedEffects.__shadow ~= nil then Lighting.GlobalShadows = BoosterSavedEffects.__shadow end
            if BoosterSavedEffects.__fog ~= nil then Lighting.FogEnd = BoosterSavedEffects.__fog end
            if BoosterSavedEffects.__quality ~= nil then settings().Rendering.QualityLevel = BoosterSavedEffects.__quality end
        end)
        for obj, was in pairs(BoosterSavedEffects) do
            if typeof(obj) == "Instance" and obj.Parent then
                pcall(function() obj.Enabled = was end)
            end
        end
        table.clear(BoosterSavedEffects)
    end
end

local function hideEntityVisualsIn(container)
    if not container then return end
    for _, desc in ipairs(container:GetDescendants()) do
        if desc:IsA("BasePart") or desc:IsA("Decal") or desc:IsA("Texture") then
            if HiddenPartsCache[desc] == nil then HiddenPartsCache[desc] = desc.Transparency end
            desc.Transparency = 1
        elseif desc:IsA("ParticleEmitter") or desc:IsA("Trail") or desc:IsA("Beam") then
            if HiddenPartsCache[desc] == nil then HiddenPartsCache[desc] = desc.Enabled end
            desc.Enabled = false
        end
    end
end

local function setEntityVisualHider(enabled)
    if enabled then
        local entities = Workspace:FindFirstChild("Entities")
        if entities then hideEntityVisualsIn(entities) end
    else
        for obj, orig in pairs(HiddenPartsCache) do
            if obj and obj.Parent then
                pcall(function()
                    if type(orig) == "number" then obj.Transparency = orig
                    elseif type(orig) == "boolean" then obj.Enabled = orig end
                end)
            end
        end
        table.clear(HiddenPartsCache)
    end
end

task.spawn(function()
    while isAliveRun() do
        if State.EntityVisualHider then
            local entities = Workspace:FindFirstChild("Entities")
            if entities then hideEntityVisualsIn(entities) end
            task.wait(1)
        else
            task.wait(1.5)
        end
    end
end)

    end

    dragBar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = true
            dragStart = input.Position
            startPos = frame.Position
            dragInput = (input.UserInputType == Enum.UserInputType.Touch) and input or nil
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                    if dragInput == input then dragInput = nil end
                end
            end)
        end
    end)
    trackConn(UserInputService.InputChanged:Connect(function(input)
        if not dragging or not frame.Parent then return end
        if input == dragInput or (dragInput == nil and input.UserInputType == Enum.UserInputType.MouseMovement) then
            local delta = input.Position - dragStart
            frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end))

    minBtn.MouseButton1Click:Connect(function()
        minimized = not minimized
        content.Visible = not minimized
        frame.Size = minimized and minSize or fullSize

        local _, myRoot = getCharacterParts()
        if not myRoot then output.Text = "Character not loaded." return end
        local entities = Workspace:FindFirstChild("Entities")
        local rows = {}
        if entities then
            for _, cat in ipairs(entities:GetChildren()) do
                local list = cat:IsA("Folder") and cat:GetChildren() or { cat }

                        table.insert(rows, {

                                #attrs > 0 and table.concat(attrs, ", ") or "none")
                        })
                    end
                end
            end
        end
        table.sort(rows, function(a, b) return a.dist < b.dist end)
        local lines = {}
        for i = 1, math.min(#rows, 14) do table.insert(lines, rows[i].txt) end
        lastText = #lines > 0 and table.concat(lines, "\n\n") or "No entities found in Workspace.Entities."
        output.Text = lastText
    end)

    scanDoorsBtn.MouseButton1Click:Connect(function()
        local lines = {}
        local doors = findDoorsFolder()
        if doors then
            table.insert(lines, "Folder: " .. doors:GetFullName())
            for _, door in ipairs(doors:GetChildren()) do
                local attrs = {}
                for k, v in pairs(door:GetAttributes()) do table.insert(attrs, k .. "=" .. tostring(v)) end
                table.insert(lines, string.format("- %s (%s) | Attrs:[%s]", door.Name, door.ClassName,
                    #attrs > 0 and table.concat(attrs, ", ") or "none"))
            end
        else
            table.insert(lines, "No Doors folder found on current map.")
        end
        lastText = table.concat(lines, "\n")
        output.Text = lastText
    end)

    copyPosBtn.MouseButton1Click:Connect(function()
        local _, myRoot = getCharacterParts()
        if myRoot then
            local p = myRoot.Position
            lastText = string.format("Vector3.new(%.1f, %.1f, %.1f)", p.X, p.Y, p.Z)
            output.Text = "Current Position:\n" .. lastText
        end
    end)

    copyBtn.MouseButton1Click:Connect(function()
        if setclipboard and lastText ~= "" then
            setclipboard(lastText)
            copyBtn.Text = "Copied!"
            task.delay(1.2, function()
                if copyBtn and copyBtn.Parent then copyBtn.Text = "Copy" end
            end)
        end
    end)

    closeBtn.MouseButton1Click:Connect(function()
        sg:Destroy()
        InspectorGui = nil
    end)
end

    State.AutoDoors = false
    State.AutoFight = false
    State.AutoM1 = false
    State.AutoG = false
    State.AutoV = false
    State.AutoZ = false
    State.AutoX = false

    State.RaidVoteReplay = false
    State.BlackScreen3D = false

    ActiveTargetEntity = nil

end

local HyperiumWindow = nil

getgenv().HyperiumCleanup = function()
    stopAllLoops()
    setStatsTracker(false)
    setEntityVisualHider(false)
    applyFPSBooster(false)

    for _, conn in ipairs(Connections) do pcall(function() conn:Disconnect() end) end
    table.clear(Connections)
    for _, gui in ipairs(CreatedGuis) do
        pcall(function() if gui and gui.Parent then gui:Destroy() end end)
    end
    table.clear(CreatedGuis)

    if HyperiumWindow then pcall(function() HyperiumWindow:Close() end) end
    pcall(function()
        local parent = getGuiParent()
        for _, child in ipairs(parent:GetChildren()) do
            if child:IsA("ScreenGui") and (child.Name == "DrRay" or child.Name == "HX_FloatingStats" or child.Name == "HX_MiniInspector") then
                child:Destroy()
            end
        end
    end)
end

local LIB_URL = "[raw.githubusercontent.com](https://raw.githubusercontent.com/AZYsGithub/DrRay-UI-Library/main/DrRay.lua)"

local function loadDrRay()
    local ok, lib = pcall(function()
        return loadstring(game:HttpGet(LIB_URL))()
    end)
    if ok and type(lib) == "table" then
        return lib
    end
    error("[Hyperium] UI library failed to load from " .. LIB_URL .. " -> " .. tostring(lib))
end
